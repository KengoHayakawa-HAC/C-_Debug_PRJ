cmake_minimum_required(VERSION 3.20)
project(boss LANGUAGES CXX)
enable_testing()
add_library(boss_logic exercise.cpp)
target_include_directories(boss_logic PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(boss_demo main.cpp)
add_executable(boss_tests tests.cpp)
target_link_libraries(boss_demo PRIVATE boss_logic)
target_link_libraries(boss_tests PRIVATE boss_logic)
foreach(target boss_logic boss_demo boss_tests)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
add_test(NAME boss_storage COMMAND boss_tests storage)
add_test(NAME boss_initialization COMMAND boss_tests initialization)
