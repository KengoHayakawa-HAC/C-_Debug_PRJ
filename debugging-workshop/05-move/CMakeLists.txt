cmake_minimum_required(VERSION 3.20)
project(move LANGUAGES CXX)
enable_testing()
add_library(move_logic exercise.cpp)
target_include_directories(move_logic PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(move_demo main.cpp)
add_executable(move_tests tests.cpp)
target_link_libraries(move_demo PRIVATE move_logic)
target_link_libraries(move_tests PRIVATE move_logic)
foreach(target move_logic move_demo move_tests)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
add_test(NAME move_transfer COMMAND move_tests transfer)
add_test(NAME move_data COMMAND move_tests data)
